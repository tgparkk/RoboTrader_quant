```python
# scripts/compute_sue.py
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent.parent))

import numpy as np
import psycopg2
from config.db_config import BACKTEST_DB_CONFIG

def compute_sue(current_eps: float, prior_eps: list[float]) -> float | None:
    """SUE = (current - mean) / stddev, 최소 4개 직전 분기 필요"""
    if len(prior_eps) < 4: return None
    mean = np.mean(prior_eps)
    std = np.std(prior_eps, ddof=1)
    if std <= 0: return None
    return (current_eps - mean) / std

def build_sue_table():
    """dart_events를 순회하며 각 발표에 SUE 부여 (스키마에 EPS가 있을 때만)"""
    with psycopg2.connect(**BACKTEST_DB_CONFIG) as conn, conn.cursor() as cur:
        cur.execute("""
            CREATE TABLE IF NOT EXISTS pead_sue (
                stock_code VARCHAR(10), announce_date VARCHAR(8),
                eps DOUBLE PRECISION, sue DOUBLE PRECISION,
                PRIMARY KEY (stock_code, announce_date)
            )
        """)
        # 스키마에 따라 쿼리 수정 필요. 아래는 예시.
        cur.execute("""
            SELECT stock_code, announce_date, eps FROM dart_events
            WHERE eps IS NOT NULL ORDER BY stock_code, announce_date
        """)
        rows = cur.fetchall()
        by_stock = {}
        for code, date, eps in rows:
            by_stock.setdefault(code, []).append((date, eps))

        for code, seq in by_stock.items():
            for i, (date, eps) in enumerate(seq):
                prior = [e for _, e in seq[max(0, i-4):i]]
                sue = compute_sue(eps, prior)
                cur.execute("""
                    INSERT INTO pead_sue VALUES (%s,%s,%s,%s)
                    ON CONFLICT (stock_code, announce_date) DO UPDATE SET eps=EXCLUDED.eps, sue=EXCLUDED.sue
                """, (code, date, eps, sue))
        conn.commit()
        print(f"pead_sue: built for {len(by_stock)} stocks")

if __name__ == "__main__":
    build_sue_table()
```

```python
# scripts/pead_backtest.py
import sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).parent.parent))

import psycopg2
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
from config.db_config import BACKTEST_DB_CONFIG

def load_daily_prices(codes: list, start: str, end: str) -> pd.DataFrame:
    with psycopg2.connect(**BACKTEST_DB_CONFIG) as conn:
        placeholders = ",".join(["%s"] * len(codes))
        df = pd.read_sql(f"""
            SELECT stock_code, date, open, close FROM daily_prices
            WHERE stock_code IN ({placeholders}) AND date BETWEEN %s AND %s
            ORDER BY stock_code, date
        """, conn, params=[*codes, start.replace("-",""), end.replace("-","")])
    return df

def load_events_with_sue(start: str, end: str) -> pd.DataFrame:
    with psycopg2.connect(**BACKTEST_DB_CONFIG) as conn:
        df = pd.read_sql(f"""
            SELECT stock_code, announce_date, sue FROM pead_sue
            WHERE sue IS NOT NULL
              AND announce_date BETWEEN '{start.replace("-","")}' AND '{end.replace("-","")}'
            ORDER BY announce_date
        """, conn)
    return df

def simulate(events: pd.DataFrame, prices: pd.DataFrame, config: dict):
    """
    config: {'sue_quantile': 'top_5', 'hold_days': 20, 'slots': 3}
    """
    # SUE 분위 필터
    q = config["sue_quantile"]
    if q == "top_5": threshold = events["sue"].quantile(0.95); filtered = events[events["sue"] >= threshold]
    elif q == "top_10": threshold = events["sue"].quantile(0.90); filtered = events[events["sue"] >= threshold]
    elif q == "bottom_5": threshold = events["sue"].quantile(0.05); filtered = events[events["sue"] <= threshold]
    elif q == "bottom_10": threshold = events["sue"].quantile(0.10); filtered = events[events["sue"] <= threshold]
    else: filtered = events

    trades = []
    price_map = prices.set_index(["stock_code", "date"])

    for _, ev in filtered.iterrows():
        code = ev["stock_code"]
        ann_date = ev["announce_date"]
        ann_dt = datetime.strptime(ann_date, "%Y%m%d")
        buy_dt = ann_dt + timedelta(days=1)
        exit_dt = buy_dt + timedelta(days=config["hold_days"])
        buy_ds = buy_dt.strftime("%Y%m%d")
        exit_ds = exit_dt.strftime("%Y%m%d")

        try:
            buy_price = price_map.loc[(code, buy_ds), "open"]
            exit_price = price_map.loc[(code, exit_ds), "close"]
        except KeyError:
            continue

        ret = (exit_price / buy_price - 1) * 100
        direction = 1 if q.startswith("top") else -1  # bottom은 숏
        trades.append({
            "code": code, "ann_date": ann_date, "buy_date": buy_ds,
            "exit_date": exit_ds, "ret": ret * direction, "sue": ev["sue"],
        })

    if not trades: return {"trades": 0, "win_rate": 0, "mean_ret": 0, "sharpe": 0}
    rets = [t["ret"] for t in trades]
    return {
        "trades": len(trades),
        "win_rate": sum(1 for r in rets if r > 0) / len(rets),
        "mean_ret": np.mean(rets),
        "total_ret": sum(rets),
        "sharpe": np.mean(rets) / (np.std(rets) + 1e-9) * np.sqrt(252 / config["hold_days"]),
    }
```

```python
# scripts/pead_backtest.py 에 추가
import itertools, json

def run_multiverse():
    start, end = "2023-01-01", "2026-03-31"
    events = load_events_with_sue(start, end)
    if len(events) < 100:
        print(f"WARN: 이벤트 수 {len(events)}개로 통계 검정력 제한")
    codes = events["stock_code"].unique().tolist()
    prices = load_daily_prices(codes, start, end)

    results = []
    for sue_q, hold, slots in itertools.product(
        ["top_5", "top_10", "bottom_5", "bottom_10"],
        [5, 10, 20, 40],
        [2, 3, 5],
    ):
        cfg = {"sue_quantile": sue_q, "hold_days": hold, "slots": slots}
        stats = simulate(events, prices, cfg)
        results.append({"config": cfg, **stats})
        print(f"{sue_q} hold={hold} slots={slots}: n={stats['trades']} wr={stats['win_rate']:.1%} mean={stats['mean_ret']:+.2f}%")

    out = Path("docs/superpowers/reports/pead-results.json")
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(results, indent=2, ensure_ascii=False, default=str))
    return results

def write_report():
    data = json.loads(Path("docs/superpowers/reports/pead-results.json").read_text())
    data.sort(key=lambda x: -x.get("sharpe", 0))
    lines = ["# PEAD 멀티버스 결과\n"]
    lines.append("| SUE 분위 | 보유일 | 슬롯 | 거래수 | 승률 | 평균수익률 | 샤프 |")
    lines.append("|---------|--------|------|--------|------|------------|------|")
    for r in data[:20]:
        c = r["config"]
        lines.append(f"| {c['sue_quantile']} | {c['hold_days']} | {c['slots']} | {r['trades']} | {r['win_rate']:.1%} | {r['mean_ret']:+.2f}% | {r.get('sharpe',0):.2f} |")
    Path("docs/superpowers/reports/pead-result.md").write_text("\n".join(lines))

if __name__ == "__main__":
    run_multiverse()
    write_report()
```
